cmake_minimum_required(VERSION 3.18)
project(NetPathBench VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(netpath-bench src/netpath_bench.cpp)
if(MSVC)
  target_compile_options(netpath-bench PRIVATE /W4 /permissive-)
else()
  target_compile_options(netpath-bench PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(WIN32)
  target_link_libraries(netpath-bench PRIVATE ws2_32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(netpath-bench PRIVATE Threads::Threads)
endif()
enable_testing()
add_test(NAME unit COMMAND netpath-bench selftest)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME loopback_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:netpath-bench>)
  set_tests_properties(loopback_integration PROPERTIES TIMEOUT 30)
endif()
